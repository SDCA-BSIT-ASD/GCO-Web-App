import { getCookie } from 'h3'

export default defineNuxtRouteMiddleware(async (to) => {
  if (to.path.startsWith('/api') || to.path === '/login') {
    return
  }

  const event = useRequestEvent()
  const requestCookie = event ? getCookie(event, 'gco_session') : null

  if (requestCookie) {
    return
  }

  if (import.meta.client) {
    try {
      await $fetch('/api/auth/session')
      return
    } catch {
      // fall through to login redirect below
    }
  }

  return navigateTo(`/login?redirect=${encodeURIComponent(to.fullPath)}`)
})

    programName.value = ''

      code: '',
      name: '',
      submit: ''
    }
  }

  function formatProgramCode(value: string) {
    return value.toUpperCase()
  }

  function validateProgramCode() {
    const value = programCode.value.trim()

    if (!value) {
      errors.value.code = 'Program code is required.'
      return false
    }

    if (!/^[A-Z0-9]+$/.test(value)) {
      errors.value.code =

      errors.value.name = 'Program name is required.'
      return false

    if (value.length < 3) {
      errors.value.name = 'Program name must be at least 3 characters long.'
      return false
    }

    errors.value.name = ''
    return true

import { createError, defineEventHandler, type H3Event, readBody } from 'h3'

import { AcademicProgramModel } from '../models/AcademicProgram'
import { UserModel } from '../models/User'
import { getSessionUser } from '../utils/auth'
import { validateAcademicProgramInput } from '../utils/academic-program'

function escapeRegex(value: string) {
  return value.replace(/[.*+?^${}()|[\]\\]/g, '\\$&')
}

async function createAcademicProgram(event: H3Event) {
  const sessionUser = getSessionUser(event)

  if (!sessionUser?.emailAddress) {
    throw createError({
      statusCode: 401,
      statusMessage: 'Unauthorized'
    })
  }

  const employee = await UserModel.findOne({
    emailAddress: sessionUser.emailAddress,
    isAdmin: true
  })

  if (!employee) {
    throw createError({
      statusCode: 403,
      statusMessage: 'Forbidden: Admin access required.'
    })
  }

  const body = await readBody(event)
  const payload = validateAcademicProgramInput(body)

  const existingProgram = await AcademicProgramModel.findOne({
    $or: [
      { programCode: { $regex: new RegExp(`^${escapeRegex(payload.programCode)}$`, 'i') } },
      { programName: { $regex: new RegExp(`^${escapeRegex(payload.programName)}$`, 'i') } }
    ]
  })

  if (existingProgram) {
    throw createError({
      statusCode: 409,
      statusMessage: 'An academic program already exists with those details.'
    })
  }

  try {
    const academicProgram = await AcademicProgramModel.create({
      programCode: payload.programCode.toUpperCase(),
      programName: payload.programName,
      isActive: payload.isActive
    })

    return {
      message: 'Academic program created successfully.',
      data: academicProgram
    }
  } catch (error) {
    if (error instanceof Error && error.name === 'ValidationError') {
      throw createError({
        statusCode: 400,
        statusMessage: error.message
      })
    }

    if (error instanceof Error && 'code' in error && error.code === 11000) {
      throw createError({
        statusCode: 409,
        statusMessage: 'An academic program already exists with those details.'
      })
    }

    throw error
  }
}

export default defineEventHandler(createAcademicProgram)

import { UserModel } from '../../../models/User'
import {
  getSessionUser,
  isAllowedEmail,
  normalizeAllowedDomains,
  setSessionUser
} from '../../../utils/auth'

export default defineEventHandler(async (event) => {
  const runtimeConfig = useRuntimeConfig(event)
  const query = getQuery(event)
  const error = query.error ? String(query.error) : ''

  if (error) {
    return sendRedirect(event, '/login?error=google-oauth-error')
  }

  const code = query.code ? String(query.code) : ''
  if (!code) {
    throw createError({
      statusCode: 400,
      statusMessage: 'Missing OAuth code.'
    })
  }

  const clientId = runtimeConfig.google?.clientId
  const clientSecret = runtimeConfig.google?.clientSecret
  const redirectUri = new URL('/api/auth/google/callback', getRequestURL(event)).toString()

  if (!clientId || !clientSecret) {
    throw createError({
      statusCode: 500,
      statusMessage: 'Google OAuth credentials are not configured.'
    })
  }

  const tokenResponse = await fetch('https://oauth2.googleapis.com/token', {
    method: 'POST',
    headers: {
      'Content-Type': 'application/x-www-form-urlencoded'
    },
    body: new URLSearchParams({
      code,
      client_id: clientId,
      client_secret: clientSecret,
      redirect_uri: redirectUri,
      grant_type: 'authorization_code'
    }).toString()
  })

  const tokenData = await tokenResponse.json().catch(() => ({}))
  if (!tokenResponse.ok || !tokenData.access_token) {
    throw createError({
      statusCode: 401,
      statusMessage: 'Unable to exchange Google OAuth authorization code.'
    })
  }

  const userInfoResponse = await fetch('https://www.googleapis.com/oauth2/v3/userinfo', {
    headers: {
      Authorization: `Bearer ${tokenData.access_token}`
    }
  })

  const googleUser = await userInfoResponse.json().catch(() => ({}))
  const emailAddress = String(googleUser.email || '').trim().toLowerCase()

  if (!emailAddress) {
    return sendRedirect(event, '/login?error=missing-email')
  }

  const allowedDomains = normalizeAllowedDomains(runtimeConfig.google?.allowedDomains || 'school.edu.ph')
  if (!isAllowedEmail(emailAddress, allowedDomains.join(','))) {
    return sendRedirect(event, '/login?error=domain-not-allowed')
  }

  const existingUser = await UserModel.findOne({ emailAddress })
  const firstName = String(googleUser.given_name || '').trim() || 'Google'
  const lastName = String(googleUser.family_name || '').trim() || 'User'

  const mappedUser = existingUser || new UserModel({
    emailAddress,
    firstName,
    lastName
  })

  mappedUser.emailAddress = emailAddress
  mappedUser.firstName = firstName
  mappedUser.lastName = lastName
  mappedUser.googleId = String(googleUser.sub || mappedUser.googleId || '')
  mappedUser.avatarUrl = String(googleUser.picture || mappedUser.avatarUrl || '')

  await mappedUser.save()

  setSessionUser(event, {
    emailAddress,
    firstName,
    lastName,
    googleId: String(googleUser.sub || ''),
    avatarUrl: String(googleUser.picture || '')
  })

  return sendRedirect(event, '/')
})

export const PROGRAM_CODE_PATTERN = /^[A-Za-z0-9]+$/
export const PROGRAM_NAME_PATTERN = /^[A-Za-z0-9]+(?: [A-Za-z0-9]+)*$/

export function validateAcademicProgramInput(payload) {
  console.log(payload)
  if (!payload || typeof payload !== 'object' || Array.isArray(payload)) {
    throw new Error('Academic program payload is required.')
  }

  const programCode = typeof payload.programCode === 'string' ? payload.programCode.trim() : ''
  const programName = typeof payload.programName === 'string' ? payload.programName.trim() : ''

  if (!programCode) {
    throw new Error('programCode is required.')
  }

  if (!programName) {
    throw new Error('programName is required.')
  }

  if (!PROGRAM_CODE_PATTERN.test(programCode)) {
    throw new Error('programCode must contain only alphanumeric characters with no spaces or special characters.')
  }

  if (!PROGRAM_NAME_PATTERN.test(programName)) {
    throw new Error('programName must contain only alphanumeric characters and spaces, with no special characters.')
  }

  return {
    programCode,
    programName,
    isActive: typeof payload.isActive === 'boolean' ? payload.isActive : true
  }
}

import crypto from 'node:crypto'
import { getCookie, setCookie } from 'h3'

export const SESSION_COOKIE = 'gco_session'

function readRuntimeConfig() {
  if (typeof useRuntimeConfig === 'function') {
    try {
      return useRuntimeConfig()
    } catch {
      return {}
    }
  }

  return {}
}

export function normalizeAllowedDomains(rawDomains = '') {
  return [...new Set(
    String(rawDomains)
      .split(',')
      .map((domain) => domain.trim().toLowerCase())
      .filter(Boolean)
  )]
}

export function isAllowedEmail(email, rawDomains = '') {
  const normalizedEmail = String(email || '').trim().toLowerCase()
  if (!normalizedEmail || !normalizedEmail.includes('@')) {
    return false
  }

  const runtimeConfig = readRuntimeConfig()
  const resolvedDomains = rawDomains || runtimeConfig.google?.allowedDomains || process.env.NUXT_ALLOWED_EMAIL_DOMAINS || 'school.edu.ph'
  const allowedDomains = normalizeAllowedDomains(resolvedDomains)

  if (allowedDomains.length === 0) {
    return false
  }

  const domain = normalizedEmail.split('@')[1]

  return allowedDomains.some((allowedDomain) => {
    return domain === allowedDomain || domain.endsWith(`.${allowedDomain}`)
  })
}

export function getSessionSecret() {
  const runtimeConfig = readRuntimeConfig()
  return process.env.NUXT_SESSION_SECRET || runtimeConfig.sessionSecret || 'gco-dev-session-secret'
}

export function encodeSessionValue(payload) {
  const raw = JSON.stringify(payload)
  const signature = crypto
    .createHmac('sha256', getSessionSecret())
    .update(raw)
    .digest('hex')

  return `${Buffer.from(raw).toString('base64url')}.${signature}`
}

export function decodeSessionValue(value) {
  if (!value || typeof value !== 'string') {
    return null
  }

  const [encoded, signature] = value.split('.')
  if (!encoded || !signature) {
    return null
  }

  const raw = Buffer.from(encoded, 'base64url').toString('utf8')
  const expectedSignature = crypto
    .createHmac('sha256', getSessionSecret())
    .update(raw)
    .digest('hex')

  if (expectedSignature !== signature) {
    return null
  }

  try {
    return JSON.parse(raw)
  } catch {
    return null
  }
}

export function setSessionUser(event, user) {
  setCookie(event, SESSION_COOKIE, encodeSessionValue(user), {
    path: '/',
    httpOnly: true,
    sameSite: 'lax',
    secure: process.env.NODE_ENV === 'production',
    maxAge: 60 * 60 * 24 * 7
  })
}

export function getSessionUser(event) {
  const value = getCookie(event, SESSION_COOKIE)
  if (!value) {
    return null
  }

  return decodeSessionValue(value)
}

  if (totalScore <= 9) return "Mild Depression";

  const byQuestion = new Map<number, number>();

  rawAnswers.forEach((item, index) => {
    if (typeof item === "number") {
      const qNum = index + 1;
      if (qNum >= 1 && qNum <= 9) {
        byQuestion.set(qNum, item);
      }
    } else if (item && typeof item === "object") {
      const record = item as Record<string, unknown>;
      const qNum = Number(record.questionNumber ?? index + 1);
      const val = Number(record.value ?? record.score ?? record.answer ?? 0);
      if (qNum >= 1 && qNum <= 9) {
        byQuestion.set(qNum, val);
      }
    }
  });

  return Array.from({ length: 9 }, (_, i) => {
    const questionNumber = i + 1;
    const rawValue = byQuestion.get(questionNumber) ?? 0;
    const value = Number.isFinite(rawValue)
      ? Math.min(3, Math.max(0, Math.round(rawValue)))
      : 0;

    return {
      questionNumber,
      statement: PHQ9_QUESTIONS[questionNumber],
      value,
      label: PHQ9_RESPONSE_LABELS[value] ?? "Not at all",
    };
  });
}

export function formatPhq9EntryDetail(params: {
  submission: {
    _id: unknown;
    submittedAt?: Date | string;
    answers?: unknown[];
  };
  student: {
    firstName?: string;
    lastName?: string;
    studentNumber?: string;
    programName?: string;
    yearLevel?: number | string;
  } | null;
}) {
  const { submission, student } = params;
  const responses = normalizePhq9Responses(submission.answers ?? []);
  const totalScore = responses.reduce((sum, item) => sum + item.value, 0);
  const severityLabel = getPhq9SeverityLabel(totalScore);

  const question9 = responses.find((item) => item.questionNumber === 9) ?? {
    questionNumber: 9,
    statement: PHQ9_QUESTIONS[9],
    value: 0,
    label: PHQ9_RESPONSE_LABELS[0],
  };

  const isFlagged = question9.value > 0;
  const flagDetails = isFlagged
    ? {
        questionNumber: 9,
        value: question9.value,
        label: question9.label,
      }
    : null;

  const fullName = [student?.firstName, student?.lastName]
    .filter(Boolean)
    .join(" ")
    .trim();

  const parsedYearLevel =
    student?.yearLevel !== undefined && student?.yearLevel !== null
      ? Number(student.yearLevel)
      : 0;

  return {
    id: String(submission._id),
    submittedAt: submission.submittedAt
      ? new Date(submission.submittedAt).toISOString()
      : new Date().toISOString(),
    student: {
      name: fullName,
      studentNumber: String(student?.studentNumber ?? ""),
      program: String(student?.programName ?? ""),
      yearLevel: Number.isFinite(parsedYearLevel) ? parsedYearLevel : 0,
    },
    responses,
    totalScore,
    severityLabel,
    isFlagged,
    flagDetails,
  };
}
